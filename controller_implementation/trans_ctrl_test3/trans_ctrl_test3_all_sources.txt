inline CanDiagnostics::Snapshot snapshot(bool mailboxes=false) {
  CanDiagnostics::Snapshot s;
  s.us=micros()-CanDiagnostics::origin; s.queue=can.getTXQueueCount();
  s.esr=FLEXCANb_ESR1(busBase); s.ecr=FLEXCANb_ECR(busBase);
  // FIFO disabled with 16 MB: only MB8..15 are TX. Never lock RX CS registers.
  if(mailboxes) for(unsigned i=8;i<16;++i) {
    const unsigned code=FLEXCAN_get_code(FLEXCANb_MBn_CS(busBase,i));
    if(code==FLEXCAN_MB_CODE_TX_INACTIVE) ++s.freeMB;
    if(code==FLEXCAN_MB_CODE_TX_ONCE) ++s.pendingMB;
  }
  CanDiagnostics::queueMax=max(CanDiagnostics::queueMax,s.queue);
  CanDiagnostics::observe(s.esr,s.ecr); return s;
}
inline bool writeDiagnosed(const CAN_message_t &m,float torque,uint8_t kind) {
  using namespace CanDiagnostics;
  const uint32_t q=can.getTXQueueCount(); queueMax=max(queueMax,q);
  const uint32_t toWrite=measuring ? uint32_t(micros()-entry):0;
  const int result=can.write(m);
  queueMax=max(queueMax,can.getTXQueueCount());
  if(result) ++success; else {
    ++failures;
    if(!firstFailure.valid) {
      failureInIsr=measuring; firstFailure.valid=true; firstFailure.kind=kind; firstFailure.torque=torque;
      firstFailure.bus=snapshot(true); firstFailure.failures=failures;
      firstFailure.age=feedback.valid ? uint32_t(micros()-feedback.rxUs):UINT32_MAX;
      firstFailure.interval=interval; firstFailure.previousDuration=duration;
      firstFailure.toWrite=toWrite;
    }
  }
  return result!=0;
}

struct ReceivedFrame { CAN_message_t frame; uint32_t rxUs; };
static ReceivedFrame received[ControlConfig::RX_QUEUE_ROWS];
static volatile uint32_t rxHead=0, rxTail=0;
static bool capture=false, haveRxTime=false;
static uint32_t previousRxUs=0;
inline void resetRx(bool enable) {
  rxHead=rxTail=0; capture=enable; haveRxTime=false; previousRxUs=0;
}
inline void receiveFeedback(const CAN_message_t &m) {
  const uint32_t now=micros();
  if(!capture) return;
  using namespace CanDiagnostics;
  ++rxFrames;
  if(m.flags.overrun) ++rxHardwareOverruns;
  if(m.flags.remote || m.flags.extended || m.len!=8 || m.buf[0]!=ControlConfig::MOTOR_ID) ++rxRejected;
  else {
    ++rxAccepted;
    if(haveRxTime) {
      const uint32_t gap=now-previousRxUs;
      rxGapMin=min(rxGapMin,gap); rxGapMax=max(rxGapMax,gap);
      if(gap<ControlConfig::RX_INTERVAL_MIN_US || gap>ControlConfig::RX_INTERVAL_MAX_US) ++rxAbnormal;
    }
    haveRxTime=true; previousRxUs=now;
    if(rxHead-rxTail>=ControlConfig::RX_QUEUE_ROWS) ++rxOverflow;
    else {
      auto &r=received[rxHead%ControlConfig::RX_QUEUE_ROWS]; r.frame=m; r.rxUs=now;
      asm volatile("" ::: "memory"); ++rxHead;
      rxQueueMax=max(rxQueueMax,uint32_t(rxHead-rxTail));
    }
  }
  rxIsrMax=max(rxIsrMax,uint32_t(micros()-now));
}
inline bool processFeedback(uint32_t now) {
  if(rxTail==rxHead) return false;
  const ReceivedFrame r=received[rxTail%ControlConfig::RX_QUEUE_ROWS];
  asm volatile("" ::: "memory"); ++rxTail;
  const auto &m=r.frame;
  using namespace ControlConfig;
  if(uint32_t(now-r.rxUs)>RX_STALE_US) ++CanDiagnostics::rxStale;
  if(feedback.valid && uint32_t(r.rxUs-feedback.rxUs)>FEEDBACK_TIMEOUT_US) communicationFault=true;
  feedback.position=decode((uint32_t(m.buf[1])<<8)|m.buf[2],POSITION_MIN,POSITION_MAX,16);
  feedback.velocity=decode((uint32_t(m.buf[3])<<4)|(m.buf[4]>>4),VELOCITY_MIN,VELOCITY_MAX,12);
  feedback.torque=decode(((uint32_t(m.buf[4])&15)<<8)|m.buf[5],TORQUE_MIN,TORQUE_MAX,12);
  feedback.error=m.buf[7]; motorFault=motorFault || feedback.error!=0;
  numericFault=numericFault || !isfinite(feedback.position) || !isfinite(feedback.velocity) || !isfinite(feedback.torque);
  feedback.rxUs=r.rxUs; ++feedback.sequence; feedback.valid=true;
  ++CanDiagnostics::rxProcessed;
  if(feedbackHandler) feedbackHandler(feedback);
  return true;
}

  can.setMaxMB(16); can.setMB(MB0,RX,STD);
  can.setMBFilter(REJECT_ALL); can.setMBFilter(MB0,ACCEPT_ALL);
  can.onReceive(MB0,receiveFeedback); can.enableMBInterrupt(MB0);

}

#pragma once
#include <stdint.h>
#include <math.h>
#include "control_config.h"
namespace ControlAlgorithm {
struct Signals {
  float velocity=0, friction=0, input=0, state=0, external=0;
  float torqueR=0, torqueF=0, unsaturated=0, command=0;
  bool saturated=false;
};
class Controller {
 public:
  Signals signals;
  float torqueBias=0;
  // Explicit new-run initialization is the only way to clear a numeric fault.
  void reset() { signals=Signals{}; initialized=false; numericFault=false; lastUs=0; }
  bool ready() const { return initialized && !numericFault; }
  bool faulted() const { return numericFault; }
  void zeroOutput() { signals.torqueR=signals.torqueF=signals.unsaturated=signals.command=0; signals.saturated=false; }
  void trip() { numericFault=true; zeroOutput(); }
  static float friction(float w) {
    if(!isfinite(w)) return NAN;
    const auto &p=w>0 ? ControlConfig::POSITIVE : ControlConfig::NEGATIVE;
    return p.tc*tanhf(w/ControlConfig::OMEGA_0)+p.b*w;
  }
  bool finiteState() const {
    return isfinite(torqueBias) && isfinite(signals.velocity) && isfinite(signals.friction) &&
      isfinite(signals.input) && isfinite(signals.state) && isfinite(signals.external) &&
      isfinite(signals.torqueR) && isfinite(signals.torqueF) && isfinite(signals.unsaturated) && isfinite(signals.command);
  }
  // Exactly once per received frame, including while torque output is OFF.
  void feedback(float rawVelocity,float motorTorque,uint32_t rxUs) {
    using namespace ControlConfig;
    if(numericFault) return;
    if(!isfinite(rawVelocity) || !isfinite(motorTorque) || !finiteState()) { trip(); return; }
    if(initialized && uint32_t(rxUs-lastUs)==0) return; // No observer update for dt=0.
    if(!initialized) {
      signals.velocity=rawVelocity;
      signals.friction=friction(signals.velocity);
      signals.input=motorTorque-torqueBias-signals.friction+J*G*signals.velocity;
      signals.state=J*G*signals.velocity; // First external estimate is exactly zero.
      initialized=true;
    } else {
      const float dt=float(uint32_t(rxUs-lastUs))*US_TO_SECONDS;
      signals.velocity+=(1.0f-expf(-RADIANS_PER_REVOLUTION*VELOCITY_HZ*dt))*(rawVelocity-signals.velocity);
      signals.friction=friction(signals.velocity);
      signals.input=motorTorque-torqueBias-signals.friction+J*G*signals.velocity;
      signals.state+=(1.0f-expf(-G*dt))*(signals.input-signals.state);
    }
    lastUs=rxUs;
    signals.external=J*G*signals.velocity-signals.state;
    if(!finiteState()) trip();
  }
  float command(bool enabled,float Kr,float Kf) {
    if(numericFault) { zeroOutput(); return 0; }
    if(!isfinite(Kr) || !isfinite(Kf) || !finiteState()) { trip(); return 0; }
    if(!enabled || !ready()) { zeroOutput(); return 0; }
    signals.torqueR=Kr*signals.external;
    signals.torqueF=Kf*signals.friction;
    signals.unsaturated=signals.torqueR+signals.torqueF;
    // Never send NaN/Inf through saturation: latch and require explicit reset.
    if(!finiteState()) { trip(); return 0; }
    signals.command=fmaxf(-ControlConfig::TORQUE_LIMIT,fminf(ControlConfig::TORQUE_LIMIT,signals.unsaturated));
    signals.saturated=signals.command!=signals.unsaturated;
    if(!isfinite(signals.command)) trip();
    return signals.command;
  }
 private:
  bool initialized=false, numericFault=false;
  uint32_t lastUs=0;
};
}

"""Convert TCBIN2 microSD logs to the existing 29-column CSV (no dependencies)."""
import argparse
import csv
from pathlib import Path
import struct
import zlib

RECORD = struct.Struct('<I10I4f9f?3x2f2B2xII')
assert RECORD.size == 120
HEADER = ('timestamp_us,position_rad,velocity_raw_rad_s,velocity_filtered_rad_s,'
          'feedback_torque_Nm,torque_bias_Nm,feedback_age_us,friction_model_Nm,'
          'rfob_input_Nm,rfob_filter_state_Nm,external_torque_est_Nm,torque_RFOB_Nm,'
          'torque_FF_Nm,torque_cmd_unsat_Nm,torque_cmd_Nm,K_R,K_F,control_enabled,'
          'torque_saturated,motor_error,fault_code,tx_queue_count,isr_interval_us,'
          'previous_isr_duration_us,can_snapshot_time_us,can_esr1,can_ecr,'
          'can_write_success,can_write_failure')


def decode(raw, expected):
    if len(raw) != RECORD.size:
        raise ValueError(f'truncated record at sequence {expected}')
    if zlib.crc32(raw[:-4]) != struct.unpack_from('<I', raw, 116)[0]:
        raise ValueError(f'CRC mismatch at sequence {expected}')
    v = RECORD.unpack(raw)
    if v[0] == 0xffffffff:
        if v[1] != expected:
            raise ValueError('end record row count mismatch')
        return None
    if v[0] != expected:
        raise ValueError(f'sequence mismatch: expected {expected}, got {v[0]}')
    # v: seq, ten uint32, four raw floats, nine signal floats, saturation,
    # Kr/Kf, enabled/error, fault, crc. Preserve legacy CSV order/units.
    return [v[1], v[11], v[12], v[15], v[13], v[14], v[2],
            *v[16:24], v[25], v[26], v[27], int(v[24]), v[28], v[29],
            *v[3:11]]


def convert(source, destination=None, recover=False):
    source = Path(source)
    destination = Path(destination) if destination else source.with_suffix('.csv')
    if source.resolve() == destination.resolve():
        raise ValueError('input and output must differ')
    # Never leave a damaged log looking like a completed CSV.

namespace PositionDirection {
enum Direction { UNKNOWN, POSITIVE, NEGATIVE };
class Estimator {
 public:
  void reset() { head=0; count=0; result=UNKNOWN; }
  Direction direction() const { return result; }
  void update(float position,uint32_t rxUs) {
    using namespace ControlConfig;
    if(count && rxUs==samples[(head+count-1)%CAPACITY].us) return;
    // Keep the latest received sample at or before now - 50 ms, plus newer samples.
    while(count>1 && uint32_t(rxUs-samples[(head+1)%CAPACITY].us)>=POSITION_DIRECTION_WINDOW_US) {
      head=(head+1)%CAPACITY; --count;
    }
    // Capacity exceeds the number of accepted frames possible in 50 ms at 1 Mbit/s.
    // If exhausted, discard incomplete history and require a fresh full window.
    if(count==CAPACITY) reset();
    samples[(head+count)%CAPACITY]={position,rxUs}; ++count;
    result=UNKNOWN;
    if(uint32_t(rxUs-samples[head].us)<POSITION_DIRECTION_WINDOW_US) return;
    float delta=position-samples[head].position;
    const float range=POSITION_MAX-POSITION_MIN;
    if(delta>range*0.5f) delta-=range;
    else if(delta<-range*0.5f) delta+=range;
    if(delta>POSITION_DIRECTION_THRESHOLD_RAD) result=POSITIVE;
    else if(delta<-POSITION_DIRECTION_THRESHOLD_RAD) result=NEGATIVE;
  }
 private:
  static constexpr uint32_t CAPACITY=1024;
  struct Sample { float position; uint32_t us; };
  Sample samples[CAPACITY];
  uint32_t head=0, count=0;
  Direction result=UNKNOWN;
};

    if(w==0) return 0;
    const auto &p=w>0 ? ControlConfig::POSITIVE : ControlConfig::NEGATIVE;
    const float r=w/p.vs;
    const float magnitude=p.tc+(p.ts-p.tc)*expf(-r*r);
    return (w>0 ? magnitude : -magnitude)+p.b*w;
  }
  bool finiteState() const {
    return isfinite(torqueBias) && isfinite(signals.velocity) && isfinite(signals.friction) &&
      isfinite(signals.input) && isfinite(signals.state) && isfinite(signals.external) &&
      isfinite(signals.torqueR) && isfinite(signals.torqueF) && isfinite(signals.unsaturated) && isfinite(signals.command);
  }
  // Exactly once per received frame, including while torque output is OFF.
  void feedback(float rawVelocity,float motorTorque,uint32_t rxUs) {
    using namespace ControlConfig;
    if(numericFault) return;
    if(!isfinite(rawVelocity) || !isfinite(motorTorque) || !finiteState()) { trip(); return; }
    if(initialized && uint32_t(rxUs-lastUs)==0) return; // No observer update for dt=0.
    if(!initialized) {
      signals.velocity=rawVelocity;
      signals.friction=friction(signals.velocity);
      signals.input=motorTorque-torqueBias-signals.friction+J*G*signals.velocity;
      signals.state=J*G*signals.velocity; // First external estimate is exactly zero.
      initialized=true;
    } else {
      const float dt=float(uint32_t(rxUs-lastUs))*US_TO_SECONDS;
      signals.velocity+=(1.0f-expf(-RADIANS_PER_REVOLUTION*VELOCITY_HZ*dt))*(rawVelocity-signals.velocity);
      signals.friction=friction(signals.velocity);
      signals.input=motorTorque-torqueBias-signals.friction+J*G*signals.velocity;
      signals.state+=(1.0f-expf(-G*dt))*(signals.input-signals.state);
    }
    lastUs=rxUs;
    signals.external=J*G*signals.velocity-signals.state;
    if(!finiteState()) trip();
  }
  float command(bool enabled,float Kr,float Kf) {
    if(numericFault) { zeroOutput(); return 0; }
    if(!isfinite(Kr) || !isfinite(Kf) || !finiteState()) { trip(); return 0; }
    if(!enabled || !ready()) { zeroOutput(); return 0; }
    signals.torqueR=Kr*signals.external;
    signals.torqueF=Kf*signals.friction;
    signals.unsaturated=signals.torqueR+signals.torqueF;
    // Never send NaN/Inf through saturation: latch and require explicit reset.
    if(!finiteState()) { trip(); return 0; }
    signals.command=fmaxf(-ControlConfig::TORQUE_LIMIT,fminf(ControlConfig::TORQUE_LIMIT,signals.unsaturated));
    signals.saturated=signals.command!=signals.unsaturated;
    if(!isfinite(signals.command)) trip();
    return signals.command;
  }
 private:
  bool initialized=false, numericFault=false;
  uint32_t lastUs=0;
};
}

    # Unique values for every field make column reordering errors visible.
    values = [sequence, *range(10, 20), *[float(x) for x in range(20, 33)],

// Single source of truth for applied runtime settings; configuration contains only defaults.
struct RuntimeSettings {
  bool enabled=ControlConfig::INITIAL_ENABLED;
  float Kr=ControlConfig::INITIAL_K_R, Kf=ControlConfig::INITIAL_K_F;
};
static volatile RuntimeSettings runtime;
volatile float tau_cmd=0;
volatile bool active=false, startRequest=false;
volatile uint32_t runStart=0, statusCode=0;
static uint32_t logTick=0;
static bool timerReady=false;
static bool awaitingFirstFeedback=true;
ControlAlgorithm::Controller controller;
IntervalTimer controlTimer;

void stopRun(uint32_t reason) {
  CanProtocol::capture=false;
  active=false; startRequest=false; runtime.enabled=false; tau_cmd=0; statusCode=reason;
  controller.zeroOutput(); // Preserve observer/numeric fault until explicit S restart.
  CanDiagnostics::stopUs=micros()-CanDiagnostics::origin; CanDiagnostics::stopReason=reason;
  CanDiagnostics::stopZero=CanProtocol::command(0); CanDiagnostics::stopDisable=CanProtocol::disable();
  CanDiagnostics::latest=CanProtocol::snapshot(true);
  if(SaveData::recording) SaveData::finish(reason);
}
// CAN and timer have the same priority: each frame is processed exactly once,
// without overwriting an unprocessed frame when several arrive in one timer period.
void handleFeedback(const CanProtocol::Feedback &f) {
  if(!active || CanProtocol::motorFault || CanProtocol::communicationFault) return;
  // A first reply arriving after the startup deadline must not hide a timeout
  // by replacing the feedback timestamp before the next control interrupt.
  if(awaitingFirstFeedback && uint32_t(f.rxUs-runStart)>ControlConfig::FEEDBACK_TIMEOUT_US) {
    CanProtocol::communicationFault=true; return;
  }
  awaitingFirstFeedback=false;
  if(CanProtocol::numericFault || !isfinite(f.position)) { controller.trip(); return; }
  controller.feedback(f.velocity,f.torque,f.rxUs); // Also while runtime.enabled==false.
}
void controlISR() {
  const uint32_t now=micros();
  if(startRequest) {
    startRequest=false; runStart=now;
    // Before active, the original observer ignored startup replies. Do not feed
    // their pre-origin timestamps to the unsigned startup-deadline check.
    CanDiagnostics::rxStartupDiscarded=CanProtocol::rxHead-CanProtocol::rxTail;
    CanProtocol::rxTail=CanProtocol::rxHead;
    active=true;
  }
  if(!active) return;
  CanDiagnostics::enter(now);
  struct TimingGuard { ~TimingGuard() { CanDiagnostics::leave(); } } timingGuard;
  if(!SaveData::recording) { stopRun(4); return; }
  for(uint32_t n=0;n<ControlConfig::RX_BUDGET && CanProtocol::processFeedback(now);++n) {}
  if(CanProtocol::rxHead!=CanProtocol::rxTail) ++CanDiagnostics::rxBudgetHits;
  const auto &f=CanProtocol::feedback;
  const uint32_t age=f.valid ? uint32_t(now-f.rxUs) : uint32_t(now-runStart);
  uint32_t fault=0;
  if(CanProtocol::motorFault) fault=7;
  else if(CanDiagnostics::rxOverflow || CanDiagnostics::rxHardwareOverruns) fault=9;
  else if(CanProtocol::communicationFault || age>ControlConfig::FEEDBACK_TIMEOUT_US) fault=6;
  if(CanProtocol::numericFault || controller.faulted() ||
     (f.valid && (!isfinite(f.position)||!isfinite(f.velocity)||!isfinite(f.torque)))) {
    controller.trip(); fault=8;
  }
  tau_cmd=controller.command(runtime.enabled && f.valid && !fault,runtime.Kr,runtime.Kf);
  if(controller.faulted() || !isfinite(tau_cmd)) { controller.trip(); tau_cmd=0; fault=8; }
  const bool txOK=CanProtocol::command(tau_cmd);
  if(!txOK && CanDiagnostics::firstFailure.valid) {
    auto &fail=CanDiagnostics::firstFailure;
    fail.age=age; fail.rxPending=CanProtocol::rxHead-CanProtocol::rxTail;
    fail.rxOverflow=CanDiagnostics::rxOverflow; fail.rxMax=CanDiagnostics::rxQueueMax;
    fail.logPending=SaveData::head-SaveData::tail; fail.logDropped=CanDiagnostics::logDropped;
  }
  if(!txOK && !fault) fault=2;
  if(ControlConfig::SD_LOGGING && (logTick++%ControlConfig::LOG_DECIMATION==0 || fault)) {
    SaveData::Sample s={}; s.timestamp=now-runStart; s.age=f.valid?age:UINT32_MAX;
    s.position=f.position; s.rawVelocity=f.velocity; s.torque=f.torque; s.bias=controller.torqueBias;
    s.signals=controller.signals; s.Kr=runtime.Kr; s.Kf=runtime.Kf;
    s.txQueue=CanProtocol::can.getTXQueueCount(); s.isrInterval=CanDiagnostics::interval; s.previousIsrUs=CanDiagnostics::duration;
    s.diagUs=CanDiagnostics::latest.us; s.esr=CanDiagnostics::latest.esr; s.ecr=CanDiagnostics::latest.ecr;
    s.txSuccess=CanDiagnostics::success; s.txFailure=CanDiagnostics::failures;
    s.enabled=runtime.enabled; s.error=f.error; s.fault=fault;
    if(!SaveData::record(s) && !fault) fault=3;
  }
  if(fault) stopRun(fault);
}
void startRun() {
  if(!timerReady || active || startRequest || SaveData::recording || !SaveData::start()) return;
  noInterrupts();
  CanDiagnostics::reset(micros());
  CanProtocol::resetRx(true);
  controller.reset(); runtime.Kr=ControlConfig::INITIAL_K_R; runtime.Kf=ControlConfig::INITIAL_K_F;
  runtime.enabled=ControlConfig::INITIAL_ENABLED; tau_cmd=0; statusCode=0; logTick=0;
  awaitingFirstFeedback=true;
  CanProtocol::feedback=CanProtocol::Feedback{};
  CanProtocol::motorFault=CanProtocol::communicationFault=CanProtocol::numericFault=false;
  const bool enabled=CanProtocol::enable(); const bool zero=CanProtocol::command(0);
  if(!enabled || !zero) stopRun(2);
  else startRequest=true;
  interrupts();
}
void printStatus() {
  noInterrupts();
  const bool enabled=runtime.enabled, running=active, ready=active && controller.ready(), numeric=controller.faulted()||CanProtocol::numericFault;
  const float Kr=runtime.Kr, Kf=runtime.Kf, torque=tau_cmd;
  const uint32_t fault=statusCode;
  interrupts();
  Serial.print("STATUS,ON,"); Serial.print(enabled?1:0);
  Serial.print(",Kr,"); Serial.print(Kr,7); Serial.print(",Kf,"); Serial.print(Kf,7);
  Serial.print(",RUNNING,"); Serial.print(running?1:0); Serial.print(",OBSERVER_READY,"); Serial.print(ready?1:0);
  Serial.print(",NUMERIC_FAULT,"); Serial.print(numeric?1:0); Serial.print(",FAULT,"); Serial.print(fault);
  Serial.print(",tau_cmd_Nm,"); Serial.println(torque,7);
  if(!running && !startRequest) SaveData::writeDiagnostics(Serial);
}
void processCommand(char *line) {
  while(isspace((unsigned char)*line)) ++line;
  char *endLine=line+strlen(line);
  while(endLine>line && isspace((unsigned char)endLine[-1])) *--endLine=0;
  for(char *p=line;*p;++p) *p=toupper((unsigned char)*p);
  if(!strcmp(line,"S")) startRun(); // Explicit reinitialization/recovery after stop.
  else if(!strcmp(line,"D")) { if(!active && !startRequest) SaveData::dumpCsv(); }
  else if(!strcmp(line,"STATUS")) printStatus();
  else if(!strcmp(line,"ON")) {
    noInterrupts();
    const bool allowed=active && !statusCode && !controller.faulted() && !CanProtocol::numericFault &&
      !CanProtocol::motorFault && !CanProtocol::communicationFault;
    if(allowed) runtime.enabled=true; // Observer is NOT reset; readiness gates output in ISR.
    interrupts();
    if(!allowed) Serial.println("ERROR,ON requires running session; use S to initialize");
  } else if(!strcmp(line,"OFF")) {
    noInterrupts(); runtime.enabled=false; tau_cmd=0; controller.zeroOutput();
    if(active && !CanProtocol::command(0)) stopRun(2);
    interrupts(); // Keep observer running while OFF.
  } else if(!strcmp(line,"BIAS")) {
    noInterrupts();
    const auto &f=CanProtocol::feedback;
    const bool allowed=active && !runtime.enabled && !controller.faulted() && f.valid &&
      !CanProtocol::motorFault && !CanProtocol::communicationFault && !CanProtocol::numericFault &&
      uint32_t(micros()-f.rxUs)<=ControlConfig::FEEDBACK_TIMEOUT_US && isfinite(f.torque) &&
      fabsf(f.velocity)<ControlConfig::BIAS_STATIONARY_SPEED;
    if(allowed) { controller.torqueBias=f.torque; controller.reset(); }
    interrupts();
    if(!allowed) Serial.println("ERROR,BIAS requires healthy OFF stationary feedback");
  } else if((!strncmp(line,"KR",2)||!strncmp(line,"KF",2)) && isspace((unsigned char)line[2])) {
    char *end=nullptr; errno=0; const float value=strtof(line+3,&end);
    while(end && isspace((unsigned char)*end)) ++end;
    if(end==line+3 || !end || *end || errno==ERANGE || !isfinite(value)) {
      Serial.println("ERROR,invalid gain; finite float required"); return;
    }
    noInterrupts(); if(line[1]=='R') runtime.Kr=value; else runtime.Kf=value; interrupts();
  } else Serial.println("ERROR,unknown command");
}
void setup() {
  Serial.begin(ControlConfig::SERIAL_BAUD); CanProtocol::feedbackHandler=handleFeedback; CanProtocol::begin();
  const bool cardReady=SaveData::begin();
  controlTimer.priority(ControlConfig::ISR_PRIORITY);
  timerReady=controlTimer.begin(controlISR,ControlConfig::CONTROL_PERIOD_US);
  Serial.print("READY,ID,"); Serial.print(ControlConfig::MOTOR_ID); Serial.print(",CAN,"); Serial.print(ControlConfig::CAN_BUS);
  Serial.print(",SD,"); Serial.print(!ControlConfig::SD_LOGGING?"DISABLED":(cardReady?"OK":"ERROR")); Serial.print(",TIMER,"); Serial.println(timerReady?"OK":"ERROR");
  Serial.println("Commands: S, ON, OFF, KR <value>, KF <value>, STATUS, BIAS, X, D (newline; X immediate)");
}
void loop() {
  static char line[ControlConfig::COMMAND_BUFFER_BYTES]; static size_t used=0; static bool overflow=false;
  while(Serial.available()) {
    const char c=Serial.read();
    // Immediate stop only as a standalone command, not inside an invalid gain token.
    if((c=='x'||c=='X') && used==0 && !overflow) { noInterrupts(); stopRun(1); interrupts(); }
    else if(c=='\r'||c=='\n') {
      if(overflow) Serial.println("ERROR,command too long");
      else if(used) { line[used]=0; processCommand(line); }
      used=0; overflow=false;
    } else if(used<sizeof(line)-1 && !overflow) line[used++]=c;
    else overflow=true;
  }
  if(active && !SaveData::recording) { noInterrupts(); stopRun(4); interrupts(); }
  static uint32_t lastDiagnosticUs=0;
  if(active && uint32_t(micros()-lastDiagnosticUs)>=100000) {
    noInterrupts(); CanDiagnostics::latest=CanProtocol::snapshot(); interrupts();
    lastDiagnosticUs=micros();
  }
  CanProtocol::serviceDiagnostics();
  const uint32_t sdStart=micros();
  SaveData::service();
  CanDiagnostics::sdMaxUs=max(CanDiagnostics::sdMaxUs,uint32_t(micros()-sdStart));
}
