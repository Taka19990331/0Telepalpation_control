#include <Arduino.h>
#include <stdlib.h>
#include <ctype.h>
#include <errno.h>
#include "can_protocol.h"
#include "control_algorithm.h"
#include "control_log.h"
#if !defined(ARDUINO_TEENSY41)
#error "Select Teensy 4.1"
#endif
// Single source of truth for applied runtime settings; configuration contains only defaults.
struct RuntimeSettings {
  bool enabled=ControlConfig::INITIAL_ENABLED;
  float Kr=ControlConfig::INITIAL_K_R, Kf=ControlConfig::INITIAL_K_F;
};
static volatile RuntimeSettings runtime;
volatile float tau_cmd=0;
volatile bool active=false, startRequest=false;
volatile uint32_t runStart=0, statusCode=0;
static bool timerReady=false;
static bool awaitingFirstFeedback=true;
ControlAlgorithm::Controller controller;
IntervalTimer controlTimer;

void stopRun(uint32_t reason) {
  CanProtocol::capture=false;
  active=false; startRequest=false; runtime.enabled=false; tau_cmd=0; statusCode=reason;
  controller.zeroOutput(); // Preserve observer/numeric fault until explicit S restart.
  CanDiagnostics::stopUs=micros()-CanDiagnostics::origin; CanDiagnostics::stopReason=reason;
  CanDiagnostics::stopZero=CanProtocol::command(0); CanDiagnostics::stopDisable=CanProtocol::disable();
  CanDiagnostics::latest=CanProtocol::snapshot(true);
}
// CAN and timer have the same priority: each frame is processed exactly once,
// without overwriting an unprocessed frame when several arrive in one timer period.
void handleFeedback(const CanProtocol::Feedback &f) {
  if(!active || CanProtocol::motorFault || CanProtocol::communicationFault) return;
  // A first reply arriving after the startup deadline must not hide a timeout
  // by replacing the feedback timestamp before the next control interrupt.
  if(awaitingFirstFeedback && uint32_t(f.rxUs-runStart)>ControlConfig::FEEDBACK_TIMEOUT_US) {
    CanProtocol::communicationFault=true; return;
  }
  awaitingFirstFeedback=false;
  if(CanProtocol::numericFault || !isfinite(f.position)) { controller.trip(); return; }
  ControlLog::append(uint32_t(f.rxUs-runStart),f.position,f.velocity);
  controller.feedback(f.position,f.velocity,f.torque,f.rxUs); // Also while runtime.enabled==false.
}
void controlISR() {
  const uint32_t now=micros();
  if(startRequest) {
    startRequest=false;
    active=true;
  }
  if(!active) return;
  CanDiagnostics::enter(now);
  struct TimingGuard { ~TimingGuard() { CanDiagnostics::leave(); } } timingGuard;
  for(uint32_t n=0;n<ControlConfig::RX_BUDGET && CanProtocol::processFeedback(now);++n) {}
  if(CanProtocol::rxHead!=CanProtocol::rxTail) ++CanDiagnostics::rxBudgetHits;
  const auto &f=CanProtocol::feedback;
  const uint32_t age=f.valid ? uint32_t(now-f.rxUs) : uint32_t(now-runStart);
  uint32_t fault=0;
  if(CanProtocol::motorFault) fault=7;
  else if(CanDiagnostics::rxOverflow || CanDiagnostics::rxHardwareOverruns) fault=9;
  else if(CanProtocol::communicationFault || age>ControlConfig::FEEDBACK_TIMEOUT_US) fault=6;
  if(CanProtocol::numericFault || controller.faulted() ||
     (f.valid && (!isfinite(f.position)||!isfinite(f.velocity)||!isfinite(f.torque)))) {
    controller.trip(); fault=8;
  }
  tau_cmd=controller.command(runtime.enabled && f.valid && !fault,runtime.Kr,runtime.Kf);
  if(controller.faulted() || !isfinite(tau_cmd)) { controller.trip(); tau_cmd=0; fault=8; }
  const bool txOK=CanProtocol::command(tau_cmd);
  if(!txOK && CanDiagnostics::firstFailure.valid) {
    auto &fail=CanDiagnostics::firstFailure;
    fail.age=age; fail.rxPending=CanProtocol::rxHead-CanProtocol::rxTail;
    fail.rxOverflow=CanDiagnostics::rxOverflow; fail.rxMax=CanDiagnostics::rxQueueMax;
  }
  if(!txOK && !fault) fault=2;
  if(fault) stopRun(fault);
}
void startRun() {
  if(!timerReady || active || startRequest) return;
  noInterrupts();
  CanDiagnostics::reset(micros());
  ControlLog::reset();
  CanProtocol::resetRx(true);
  runStart=micros();
  controller.reset(); runtime.Kr=ControlConfig::INITIAL_K_R; runtime.Kf=ControlConfig::INITIAL_K_F;
  runtime.enabled=ControlConfig::INITIAL_ENABLED; tau_cmd=0; statusCode=0;
  awaitingFirstFeedback=true;
  CanProtocol::feedback=CanProtocol::Feedback{};
  CanProtocol::motorFault=CanProtocol::communicationFault=CanProtocol::numericFault=false;
  const bool enabled=CanProtocol::enable(); const bool zero=CanProtocol::command(0);
  if(!enabled || !zero) stopRun(2);
  else startRequest=true;
  interrupts();
}
void printStatus() {
  noInterrupts();
  const bool enabled=runtime.enabled, running=active, ready=active && controller.ready(), numeric=controller.faulted()||CanProtocol::numericFault;
  const float Kr=runtime.Kr, Kf=runtime.Kf, torque=tau_cmd;
  const float angleStart=controller.angleStartDeg(), angleFull=controller.angleFullDeg();
  const uint32_t fault=statusCode;
  const uint8_t notchMode=uint8_t(controller.notchMode());
  interrupts();
  Serial.print("STATUS,ON,"); Serial.print(enabled?1:0);
  Serial.print(",Kr,"); Serial.print(Kr,7); Serial.print(",Kf,"); Serial.print(Kf,7);
  Serial.print(",RUNNING,"); Serial.print(running?1:0); Serial.print(",OBSERVER_READY,"); Serial.print(ready?1:0);
  Serial.print(",NUMERIC_FAULT,"); Serial.print(numeric?1:0); Serial.print(",FAULT,"); Serial.print(fault);
  Serial.print(",NOTCH_MODE,"); Serial.print(unsigned(notchMode));
  Serial.print(",ANGLE_START_deg_RMS,"); Serial.print(angleStart,5);
  Serial.print(",ANGLE_FULL_deg_RMS,"); Serial.print(angleFull,5);
  Serial.print(",tau_cmd_Nm,"); Serial.println(torque,7);
}
void processCommand(char *line) {
  while(isspace((unsigned char)*line)) ++line;
  char *endLine=line+strlen(line);
  while(endLine>line && isspace((unsigned char)endLine[-1])) *--endLine=0;
  for(char *p=line;*p;++p) *p=toupper((unsigned char)*p);
  if(!strcmp(line,"S")) startRun(); // Explicit reinitialization/recovery after stop.
  else if(!strcmp(line,"STATUS")) printStatus();
  else if(!strcmp(line,"D")) {
    noInterrupts(); const bool allowed=!active && !startRequest; interrupts();
    if(allowed) ControlLog::dump();
    else Serial.println("ERROR,D requires a stopped control session");
  }
  else if(!strcmp(line,"ON")) {
    noInterrupts();
    const bool allowed=active && !statusCode && !controller.faulted() && !CanProtocol::numericFault &&
      !CanProtocol::motorFault && !CanProtocol::communicationFault;
    if(allowed) runtime.enabled=true; // Observer is NOT reset; readiness gates output in ISR.
    interrupts();
    if(!allowed) Serial.println("ERROR,ON requires running session; use S to initialize");
  } else if(!strcmp(line,"OFF")) {
    noInterrupts(); runtime.enabled=false; tau_cmd=0; controller.zeroOutput();
    if(active && !CanProtocol::command(0)) stopRun(2);
    interrupts(); // Keep observer running while OFF.
  } else if(!strcmp(line,"BIAS")) {
    noInterrupts();
    const auto &f=CanProtocol::feedback;
    const bool allowed=active && !runtime.enabled && !controller.faulted() && f.valid &&
      !CanProtocol::motorFault && !CanProtocol::communicationFault && !CanProtocol::numericFault &&
      uint32_t(micros()-f.rxUs)<=ControlConfig::FEEDBACK_TIMEOUT_US && isfinite(f.torque) &&
      fabsf(f.velocity)<ControlConfig::BIAS_STATIONARY_SPEED;
    if(allowed) { controller.torqueBias=f.torque; controller.reset(); }
    interrupts();
    if(!allowed) Serial.println("ERROR,BIAS requires healthy OFF stationary feedback");
  } else if(!strncmp(line,"NOTCH ",6)) {
    const bool allowed=!runtime.enabled;
    ControlConfig::NotchMode mode;
    if(!strcmp(line+6,"OFF")) mode=ControlConfig::NOTCH_OFF;
    else if(!strcmp(line+6,"FIXED")) mode=ControlConfig::NOTCH_FIXED;
    else if(!strcmp(line+6,"ADAPTIVE")) mode=ControlConfig::NOTCH_ADAPTIVE;
    else { Serial.println("ERROR,NOTCH expects OFF, FIXED, or ADAPTIVE"); return; }
    if(allowed) { noInterrupts(); controller.setNotchMode(mode); interrupts(); }
    else Serial.println("ERROR,NOTCH mode can only change while control is OFF");
  } else if(!strncmp(line,"ANGLE_START ",12)||!strncmp(line,"ANGLE_FULL ",11)) {
    const bool isStart=line[6]=='S';
    const char *arg=line+(isStart?12:11);
    char *end=nullptr; errno=0; const float value=strtof(arg,&end);
    while(end && isspace((unsigned char)*end)) ++end;
    if(end==arg || !end || *end || errno==ERANGE || !isfinite(value)) {
      Serial.println("ERROR,invalid angle threshold; finite degrees RMS required"); return;
    }
    noInterrupts();
    const float start=isStart?value:controller.angleStartDeg();
    const float full=isStart?controller.angleFullDeg():value;
    const bool changed=controller.setAngleThresholds(start,full);
    const float currentStart=controller.angleStartDeg(), currentFull=controller.angleFullDeg();
    interrupts();
    if(changed) { Serial.print("OK,ANGLE_START_deg_RMS,"); Serial.print(currentStart,5); Serial.print(",ANGLE_FULL_deg_RMS,"); Serial.println(currentFull,5); }
    else Serial.println("ERROR,require 0 < ANGLE_START < ANGLE_FULL (deg RMS); previous settings retained");
  } else if((!strncmp(line,"KR",2)||!strncmp(line,"KF",2)) && isspace((unsigned char)line[2])) {
    char *end=nullptr; errno=0; const float value=strtof(line+3,&end);
    while(end && isspace((unsigned char)*end)) ++end;
    if(end==line+3 || !end || *end || errno==ERANGE || !isfinite(value)) {
      Serial.println("ERROR,invalid gain; finite float required"); return;
    }
    noInterrupts(); if(line[1]=='R') runtime.Kr=value; else runtime.Kf=value; interrupts();
  } else Serial.println("ERROR,unknown command");
}
void setup() {
  Serial.begin(ControlConfig::SERIAL_BAUD); CanProtocol::feedbackHandler=handleFeedback; CanProtocol::begin();
  controlTimer.priority(ControlConfig::ISR_PRIORITY);
  timerReady=controlTimer.begin(controlISR,ControlConfig::CONTROL_PERIOD_US);
  Serial.print("READY,ID,"); Serial.print(ControlConfig::MOTOR_ID); Serial.print(",CAN,"); Serial.print(ControlConfig::CAN_BUS);
  Serial.print(",TIMER,"); Serial.println(timerReady?"OK":"ERROR");
  Serial.println("Commands: S, ON, OFF, KR <value>, KF <value>, NOTCH OFF|FIXED|ADAPTIVE, ANGLE_START <deg_RMS>, ANGLE_FULL <deg_RMS>, STATUS, BIAS, D (dump position/velocity log while stopped), X (newline; X immediate)");
}
void loop() {
  static char line[ControlConfig::COMMAND_BUFFER_BYTES]; static size_t used=0; static bool overflow=false;
  while(Serial.available()) {
    const char c=Serial.read();
    // Immediate stop only as a standalone command, not inside an invalid gain token.
    if((c=='x'||c=='X') && used==0 && !overflow) { noInterrupts(); stopRun(1); interrupts(); }
    else if(c=='\r'||c=='\n') {
      if(overflow) Serial.println("ERROR,command too long");
      else if(used) { line[used]=0; processCommand(line); }
      used=0; overflow=false;
    } else if(used<sizeof(line)-1 && !overflow) line[used++]=c;
    else overflow=true;
  }
  static uint32_t lastDiagnosticUs=0;
  if(active && uint32_t(micros()-lastDiagnosticUs)>=100000) {
    noInterrupts(); CanDiagnostics::latest=CanProtocol::snapshot(); interrupts();
    lastDiagnosticUs=micros();
  }
  CanProtocol::serviceDiagnostics();
}


===== can_diagnostics.h =====
#pragma once
// Fixed-size RAM diagnostics. No formatting or Serial in interrupt context.
namespace CanDiagnostics {
struct Snapshot {
  uint32_t us=0, queue=0, esr=0, ecr=0, freeMB=0, pendingMB=0;
};
struct Failure {
  bool valid=false; uint8_t kind=0; float torque=0;
  uint32_t age=0, interval=0, previousDuration=0, toWrite=0, duration=0, failures=0;
  uint32_t rxPending=0, rxOverflow=0, rxMax=0;
  Snapshot bus;
};
static uint32_t origin=0, success=0, failures=0, queueMax=0, rxGapMax=0;
static uint32_t interval=0, intervalMax=0, duration=0, durationMax=0;
static uint32_t intervalOverruns=0, executionOverruns=0, lastEntry=0, entry=0;
static uint32_t errorRecords=0, errorBits=0, txErrorMax=0, rxErrorMax=0, busOffObserved=0;
static uint32_t intervalMin=UINT32_MAX, rxGapMin=UINT32_MAX, rxIsrMax=0;
static uint32_t rxFrames=0, rxAccepted=0, rxRejected=0, rxOverflow=0, rxQueueMax=0;
static uint32_t rxHardwareOverruns=0, rxStale=0, rxAbnormal=0, rxProcessed=0, rxBudgetHits=0;
static uint32_t rxStartupDiscarded=0;
static bool measuring=false, failureInIsr=false;
static Snapshot latest;
static Failure firstFailure;
static bool stopZero=false, stopDisable=false;
static uint32_t stopUs=0, stopReason=0;
inline void reset(uint32_t now) {
  origin=now; success=failures=queueMax=rxGapMax=0;
  interval=intervalMax=duration=durationMax=intervalOverruns=executionOverruns=0;
  lastEntry=entry=0; errorRecords=errorBits=txErrorMax=rxErrorMax=busOffObserved=0;
  measuring=false; failureInIsr=false; latest=Snapshot{}; firstFailure=Failure{};
  intervalMin=rxGapMin=UINT32_MAX; rxIsrMax=0;
  rxFrames=rxAccepted=rxRejected=rxOverflow=rxQueueMax=0;
  rxHardwareOverruns=rxStale=rxAbnormal=rxProcessed=rxBudgetHits=0;
  rxStartupDiscarded=0;
  stopZero=stopDisable=false; stopUs=stopReason=0;
}
inline void observe(uint32_t esr,uint32_t ecr) {
  errorBits|=esr; txErrorMax=max(txErrorMax,ecr&255U); rxErrorMax=max(rxErrorMax,(ecr>>8)&255U);
  if((esr&0x30U)==0x20U || (esr&0x04U)) busOffObserved=1;
}
inline void enter(uint32_t now) {
  entry=now; measuring=true;
  interval=lastEntry ? uint32_t(now-lastEntry):0; lastEntry=now;
  intervalMax=max(intervalMax,interval);
  if(interval) intervalMin=min(intervalMin,interval);
  if(interval>ControlConfig::CONTROL_PERIOD_US) ++intervalOverruns;
}
inline void leave() {
  duration=micros()-entry; durationMax=max(durationMax,duration);
  if(duration>ControlConfig::CONTROL_PERIOD_US) ++executionOverruns;
  if(failureInIsr)
    firstFailure.duration=duration;
  failureInIsr=false;
  measuring=false;
}
}




===== can_protocol.h =====
#pragma once
#include <Arduino.h>
#include <math.h>
#include <FlexCAN_T4.h>
#include <IntervalTimer.h>
#include "control_config.h"
#include "can_diagnostics.h"
namespace CanProtocol {
using CanType=FlexCAN_T4<ControlConfig::CAN_BUS==1 ? CAN1 : (ControlConfig::CAN_BUS==2 ? CAN2 : CAN3),RX_SIZE_256,TX_SIZE_16>;
static CanType can;
struct Feedback { float position=0, velocity=0, torque=0; uint32_t rxUs=0, sequence=0; uint8_t error=0; bool valid=false; };
static Feedback feedback;
constexpr uint32_t busBase=ControlConfig::CAN_BUS==1 ? CAN1 : (ControlConfig::CAN_BUS==2 ? CAN2 : CAN3);
inline CanDiagnostics::Snapshot snapshot(bool mailboxes=false) {
  CanDiagnostics::Snapshot s;
  s.us=micros()-CanDiagnostics::origin; s.queue=can.getTXQueueCount();
  s.esr=FLEXCANb_ESR1(busBase); s.ecr=FLEXCANb_ECR(busBase);
  // FIFO disabled with 16 MB: only MB8..15 are TX. Never lock RX CS registers.
  if(mailboxes) for(unsigned i=8;i<16;++i) {
    const unsigned code=FLEXCAN_get_code(FLEXCANb_MBn_CS(busBase,i));
    if(code==FLEXCAN_MB_CODE_TX_INACTIVE) ++s.freeMB;
    if(code==FLEXCAN_MB_CODE_TX_ONCE) ++s.pendingMB;
  }
  CanDiagnostics::queueMax=max(CanDiagnostics::queueMax,s.queue);
  CanDiagnostics::observe(s.esr,s.ecr); return s;
}
inline bool writeDiagnosed(const CAN_message_t &m,float torque,uint8_t kind) {
  using namespace CanDiagnostics;
  const uint32_t q=can.getTXQueueCount(); queueMax=max(queueMax,q);
  const uint32_t toWrite=measuring ? uint32_t(micros()-entry):0;
  const int result=can.write(m);
  queueMax=max(queueMax,can.getTXQueueCount());
  if(result) ++success; else {
    ++failures;
    if(!firstFailure.valid) {
      failureInIsr=measuring; firstFailure.valid=true; firstFailure.kind=kind; firstFailure.torque=torque;
      firstFailure.bus=snapshot(true); firstFailure.failures=failures;
      firstFailure.age=feedback.valid ? uint32_t(micros()-feedback.rxUs):UINT32_MAX;
      firstFailure.interval=interval; firstFailure.previousDuration=duration;
      firstFailure.toWrite=toWrite;
    }
  }
  return result!=0;
}
// error() decodes strings and masks CAN IRQ internally: loop only, bounded work.
inline void serviceDiagnostics() {
  CAN_error_t e;
  for(unsigned i=0;i<4 && can.error(e,false);++i) {
    noInterrupts(); ++CanDiagnostics::errorRecords;
    CanDiagnostics::observe(e.ESR1,e.ECR); interrupts();
  }
}


static bool motorFault=false, communicationFault=false, numericFault=false;
static void (*feedbackHandler)(const Feedback &)=nullptr;
inline uint32_t encode(float x,float lo,float hi,uint8_t bits) {
  x=constrain(x,lo,hi); return uint32_t((x-lo)*float((1UL<<bits)-1)/(hi-lo));
}
inline float decode(uint32_t x,float lo,float hi,uint8_t bits) { return float(x)*(hi-lo)/float((1UL<<bits)-1)+lo; }
inline bool sendSpecial(uint8_t code) {
  CAN_message_t m={}; m.id=ControlConfig::MOTOR_ID; m.len=8;
  for(uint8_t i=0;i<7;++i) m.buf[i]=0xff;
  m.buf[7]=code; return writeDiagnosed(m,0,code);
}
inline bool enable() { return sendSpecial(0xfc); }
inline bool disable() { return sendSpecial(0xfd); }
// Only torque is configurable at runtime. MIT position/velocity/Kp/Kd stay zero.
inline bool command(float tau) {
  using namespace ControlConfig;
  if(!isfinite(tau)) { numericFault=true; tau=0; }
  const uint32_t pi=encode(0,POSITION_MIN,POSITION_MAX,16);
  const uint32_t vi=encode(0,VELOCITY_MIN,VELOCITY_MAX,12);
  const uint32_t kpi=encode(0,0,MIT_KP_MAX,12), kdi=encode(0,0,MIT_KD_MAX,12);
  const uint32_t ti=encode(tau,TORQUE_MIN,TORQUE_MAX,12);
  CAN_message_t m={}; m.id=MOTOR_ID; m.len=8;
  m.buf[0]=pi>>8; m.buf[1]=pi; m.buf[2]=vi>>4;
  m.buf[3]=((vi&15)<<4)|(kpi>>8); m.buf[4]=kpi;
  m.buf[5]=kdi>>4; m.buf[6]=((kdi&15)<<4)|(ti>>8); m.buf[7]=ti;
  return writeDiagnosed(m,tau,0);
}
// CAN and PIT have equal priority on Teensy: callbacks cannot preempt controlISR.
// Foreground resets this SPSC queue only with interrupts masked. No events().
struct ReceivedFrame { CAN_message_t frame; uint32_t rxUs; };
static ReceivedFrame received[ControlConfig::RX_QUEUE_ROWS];
static volatile uint32_t rxHead=0, rxTail=0;
static bool capture=false, haveRxTime=false;
static uint32_t previousRxUs=0;
inline void resetRx(bool enable) {
  rxHead=rxTail=0; capture=enable; haveRxTime=false; previousRxUs=0;
}
inline void receiveFeedback(const CAN_message_t &m) {
  const uint32_t now=micros();
  if(!capture) return;
  using namespace CanDiagnostics;
  ++rxFrames;
  if(m.flags.overrun) ++rxHardwareOverruns;
  if(m.flags.remote || m.flags.extended || m.len!=8 || m.buf[0]!=ControlConfig::MOTOR_ID) ++rxRejected;
  else {
    ++rxAccepted;
    if(haveRxTime) {
      const uint32_t gap=now-previousRxUs;
      rxGapMin=min(rxGapMin,gap); rxGapMax=max(rxGapMax,gap);
      if(gap<ControlConfig::RX_INTERVAL_MIN_US || gap>ControlConfig::RX_INTERVAL_MAX_US) ++rxAbnormal;
    }
    haveRxTime=true; previousRxUs=now;
    if(rxHead-rxTail>=ControlConfig::RX_QUEUE_ROWS) ++rxOverflow;
    else {
      auto &r=received[rxHead%ControlConfig::RX_QUEUE_ROWS]; r.frame=m; r.rxUs=now;
      asm volatile("" ::: "memory"); ++rxHead;
      rxQueueMax=max(rxQueueMax,uint32_t(rxHead-rxTail));
    }
  }
  rxIsrMax=max(rxIsrMax,uint32_t(micros()-now));
}
inline bool processFeedback(uint32_t now) {
  if(rxTail==rxHead) return false;
  const ReceivedFrame r=received[rxTail%ControlConfig::RX_QUEUE_ROWS];
  asm volatile("" ::: "memory"); ++rxTail;
  const auto &m=r.frame;
  using namespace ControlConfig;
  if(uint32_t(now-r.rxUs)>RX_STALE_US) ++CanDiagnostics::rxStale;
  if(feedback.valid && uint32_t(r.rxUs-feedback.rxUs)>FEEDBACK_TIMEOUT_US) communicationFault=true;
  feedback.position=decode((uint32_t(m.buf[1])<<8)|m.buf[2],POSITION_MIN,POSITION_MAX,16);
  feedback.velocity=decode((uint32_t(m.buf[3])<<4)|(m.buf[4]>>4),VELOCITY_MIN,VELOCITY_MAX,12);
  feedback.torque=decode(((uint32_t(m.buf[4])&15)<<8)|m.buf[5],TORQUE_MIN,TORQUE_MAX,12);
  feedback.error=m.buf[7]; motorFault=motorFault || feedback.error!=0;
  numericFault=numericFault || !isfinite(feedback.position) || !isfinite(feedback.velocity) || !isfinite(feedback.torque);
  feedback.rxUs=r.rxUs; ++feedback.sequence; feedback.valid=true;
  ++CanDiagnostics::rxProcessed;
  if(feedbackHandler) feedbackHandler(feedback);
  return true;
}
inline void begin() {
  can.begin(); can.setBaudRate(ControlConfig::CAN_BAUD);
  can.setMaxMB(16); can.setMB(MB0,RX,STD);
  can.setMBFilter(REJECT_ALL); can.setMBFilter(MB0,ACCEPT_ALL);
  can.onReceive(MB0,receiveFeedback); can.enableMBInterrupt(MB0);
  NVIC_SET_PRIORITY(ControlConfig::CAN_BUS==1 ? IRQ_CAN1 : (ControlConfig::CAN_BUS==2 ? IRQ_CAN2 : IRQ_CAN3),ControlConfig::ISR_PRIORITY);
}
}


===== control_algorithm.h =====
#pragma once
#include <stdint.h>
#include <math.h>
#include "control_config.h"
#include "fixed_notch.h"
namespace ControlAlgorithm {
struct Signals {
  float velocity=0, friction=0, input=0, state=0, external=0;
  float torqueR=0, torqueF=0, unsaturated=0, command=0;
  float angleBpf=0, angleRms=0, meanSpeed=0, gammaTheta=0, gammaSpeed=1, gamma=0;
  float notchInput=0, notchOutput=0;
  uint8_t notchMode=ControlConfig::INITIAL_NOTCH_MODE;
  bool saturated=false;
};
// Bilinear-transform Butterworth biquad, transposed direct-form II.
class Biquad {
 public:
  Biquad(float cutoff,float q,bool highpass) {
    const double w=6.2831853071795864769*cutoff*ControlConfig::CONTROL_PERIOD_US*1e-6;
    const double c=cos(w), s=sin(w), alpha=s/(2*q), a0=1+alpha;
    if(highpass) { b0=(1+c)/(2*a0); b1=-(1+c)/a0; b2=b0; }
    else { b0=(1-c)/(2*a0); b1=(1-c)/a0; b2=b0; }
    a1=-2*c/a0; a2=(1-alpha)/a0;
  }
  void reset() { z1=z2=0; initialized=false; }
  double update(double x) {
    if(!initialized) {
      const double dc=(b0+b1+b2)/(1+a1+a2);
      const double y=dc*x; z1=y-b0*x; z2=b2*x-a2*y; initialized=true;
    }
    const double y=b0*x+z1; z1=b1*x-a1*y+z2; z2=b2*x-a2*y; return y;
  }
  double magnitude(double w) const {
    const double zr=cos(w),zi=-sin(w),z2r=cos(2*w),z2i=-sin(2*w);
    const double nr=b0+b1*zr+b2*z2r,ni=b1*zi+b2*z2i;
    const double dr=1+a1*zr+a2*z2r,di=a1*zi+a2*z2i;
    return sqrt((nr*nr+ni*ni)/(dr*dr+di*di));
  }
  bool finiteState() const { return isfinite(z1)&&isfinite(z2); }
 private: double b0=0,b1=0,b2=0,a1=0,a2=0,z1=0,z2=0; bool initialized=false;
};
class AngleDetector {
 public:
  AngleDetector(): hp1(ControlConfig::ANGLE_BPF_LOW_HZ,0.541196100146197,true), hp2(ControlConfig::ANGLE_BPF_LOW_HZ,1.306562964876377,true),
    lp1(ControlConfig::ANGLE_BPF_HIGH_HZ,0.541196100146197,false), lp2(ControlConfig::ANGLE_BPF_HIGH_HZ,1.306562964876377,false) {
    const double fc=sqrt(double(ControlConfig::ANGLE_BPF_LOW_HZ)*ControlConfig::ANGLE_BPF_HIGH_HZ);
    const double w=6.2831853071795864769*fc*ControlConfig::CONTROL_PERIOD_US*1e-6;
    const double h=hp1.magnitude(w)*hp2.magnitude(w)*lp1.magnitude(w)*lp2.magnitude(w);
    normalization=1.0/h;
  }
  void reset() { hp1.reset();hp2.reset();lp1.reset();lp2.reset();initialized=false;unwrapped=0;rms2=meanSpeed=0; }
  void update(float position,float velocity,uint32_t dtUs) {
    using namespace ControlConfig;
    if(!initialized) { unwrapped=position; initialized=true; }
    else {
      double d=double(position)-lastPosition, span=ANGLE_WRAP_MAX_RAD-ANGLE_WRAP_MIN_RAD;
      if(d>span*.5) d-=span; else if(d< -span*.5) d+=span;
      unwrapped+=d;
    }
    lastPosition=position;
    double y=hp1.update(unwrapped); y=hp2.update(y); y=lp1.update(y); y=lp2.update(y)*normalization;
    const double dt=double(CONTROL_PERIOD_US)*1e-6;
    const double ar=exp(-dt/ANGLE_RMS_TAU_S); rms2=ar*rms2+(1-ar)*y*y;
    const double av=exp(-6.2831853071795864769*MEAN_SPEED_LPF_HZ*double(dtUs)*1e-6);
    meanSpeed=av*meanSpeed+(1-av)*double(velocity);
    bpf=float(y); rms=float(sqrt(rms2)); speed=float(fabs(meanSpeed));
  }
  bool finiteState() const { return hp1.finiteState()&&hp2.finiteState()&&lp1.finiteState()&&lp2.finiteState()&&isfinite(unwrapped)&&isfinite(rms2)&&isfinite(meanSpeed)&&isfinite(normalization); }
  float bpf=0,rms=0,speed=0;
 private:
  Biquad hp1,hp2,lp1,lp2; bool initialized=false; double lastPosition=0,unwrapped=0,rms2=0,meanSpeed=0,normalization=1;
};
class Controller {
 public:
  Signals signals;
  float torqueBias=0;
  void reset() { mode=ControlConfig::INITIAL_NOTCH_MODE; signals=Signals{}; signals.notchMode=uint8_t(mode); initialized=false; numericFault=false; lastUs=0; notch.reset(); velocityNotch.reset(); detector.reset(); latestRawVelocity=0; for(uint8_t i=0;i<ControlConfig::VELOCITY_MOVING_AVERAGE_SAMPLES;++i) velocitySamples[i]=0; velocitySum=0; velocityIndex=0; velocityCount=0; appliedGamma=0; holdSeconds=0; }
  bool ready() const { return initialized && !numericFault; }
  bool faulted() const { return numericFault; }
  void zeroOutput() { signals.torqueR=signals.torqueF=signals.unsaturated=signals.command=0; signals.saturated=false; }
  void trip() { numericFault=true; zeroOutput(); }
  void setNotchMode(ControlConfig::NotchMode m) { mode=m; signals.notchMode=uint8_t(m); appliedGamma=0; holdSeconds=0; }
  ControlConfig::NotchMode notchMode() const { return mode; }
  bool setAngleThresholds(float startDeg,float fullDeg) {
    if(!isfinite(startDeg)||!isfinite(fullDeg)||startDeg<=0||fullDeg<=startDeg) return false;
    angleStartThresholdDeg=startDeg; angleFullThresholdDeg=fullDeg; return true;
  }
  float angleStartDeg() const { return angleStartThresholdDeg; }
  float angleFullDeg() const { return angleFullThresholdDeg; }
  static float friction(float w,ControlConfig::FrictionModel model) {
    if(!isfinite(w)) return NAN;
    const auto &p=w>0 ? ControlConfig::POSITIVE : ControlConfig::NEGATIVE;
    if(model==ControlConfig::FRICTION_STRIBECK) {
      const float r=w/p.vs;
      return tanhf(w/ControlConfig::OMEGA_0)*
        (p.tc+(p.ts-p.tc)*expf(-r*r)+p.b*fabsf(w));
    }
    return p.tc*tanhf(w/ControlConfig::OMEGA_0)+p.b*w;
  }
  bool finiteState() const {
    return notch.finiteState()&&velocityNotch.finiteState()&&detector.finiteState()&&
      isfinite(torqueBias)&&isfinite(latestRawVelocity)&&isfinite(velocitySum)&&isfinite(appliedGamma)&&isfinite(holdSeconds)&&
      isfinite(signals.velocity)&&isfinite(signals.friction)&&isfinite(signals.input)&&isfinite(signals.state)&&isfinite(signals.external)&&
      isfinite(signals.torqueR)&&isfinite(signals.torqueF)&&isfinite(signals.unsaturated)&&isfinite(signals.command)&&
      isfinite(signals.angleBpf)&&isfinite(signals.angleRms)&&isfinite(signals.meanSpeed)&&isfinite(signals.gammaTheta)&&isfinite(signals.gammaSpeed)&&isfinite(signals.gamma)&&isfinite(signals.notchInput)&&isfinite(signals.notchOutput);
  }
  // Called exactly once per new CAN feedback frame; BPF is designed for 2 kHz sampling.
  void feedback(float position,float rawVelocity,float motorTorque,uint32_t rxUs) {
    using namespace ControlConfig;
    if(numericFault) return;
    if(!isfinite(position)||!isfinite(rawVelocity)||!isfinite(motorTorque)||!finiteState()) { trip(); return; }
    if(initialized && uint32_t(rxUs-lastUs)==0) return;
    const uint32_t dtUs=initialized?uint32_t(rxUs-lastUs):CONTROL_PERIOD_US;
    detector.update(position,rawVelocity,dtUs);
    constexpr float VELOCITY_DEADZONE_RAD_S=0.08f;
    const float velocityMagnitude=fabsf(rawVelocity);
    const float velocityForAverage=velocityMagnitude<=VELOCITY_DEADZONE_RAD_S ? 0.0f :
      (rawVelocity>0.0f ? 1.0f : -1.0f)*(velocityMagnitude-VELOCITY_DEADZONE_RAD_S);
    const float filteredVelocity=updateVelocityMovingAverage(velocityForAverage);
    if(!initialized) {
      signals.velocity=filteredVelocity; signals.friction=friction(signals.velocity,RFOB_FRICTION_MODEL);
      signals.input=motorTorque-torqueBias-signals.friction+J*G*signals.velocity;
      signals.state=J*G*signals.velocity; initialized=true;
    } else {
      const float dt=float(dtUs)*US_TO_SECONDS;
      signals.velocity=filteredVelocity;
      signals.friction=friction(signals.velocity,RFOB_FRICTION_MODEL);
      signals.input=motorTorque-torqueBias-signals.friction+J*G*signals.velocity;
      signals.state+=(1.0f-expf(-G*dt))*(signals.input-signals.state);
    }
    lastUs=rxUs; latestRawVelocity=rawVelocity;
    signals.external=J*G*signals.velocity-signals.state;
    signals.angleBpf=detector.bpf; signals.angleRms=detector.rms; signals.meanSpeed=detector.speed;
    if(!finiteState()) trip();
  }
  // Called once per 500 us command tick. Both notch instances run continuously.
  // notch: diagnostics only. velocityNotch: damping branch.
  float command(bool enabled,float Kr,float Kf) {
    using namespace ControlConfig;
    if(numericFault) { zeroOutput(); return 0; }
    if(!isfinite(Kr)||!isfinite(Kf)||!finiteState()) { trip(); return 0; }
    const float raw=signals.external, filtered=notch.update(raw);
    // Sample-and-hold the latest raw CAN velocity. Do not reuse the RFOB 5-point moving average.
    // The notch includes NOTCH_DEPTH internally; do not multiply depth again.
    const float velocityNotched=velocityNotch.update(latestRawVelocity);
    const float oscillationVelocity=latestRawVelocity-velocityNotched;
    const double dt=CONTROL_PERIOD_US*1e-6;
    float gt=0,gv=1;
    if(mode==NOTCH_FIXED) { gt=gv=1; }
    else if(mode==NOTCH_ADAPTIVE) {
      const float ar=angleStartThresholdDeg*0.01745329251994329577f;
      const float full=angleFullThresholdDeg*0.01745329251994329577f;
      gt=fmaxf(0,fminf(1,(signals.angleRms-ar)/(full-ar)));
      if(ENABLE_SPEED_GATE) gv=fmaxf(0,fminf(1,(SPEED_GATE_ZERO_RAD_S-signals.meanSpeed)/(SPEED_GATE_ZERO_RAD_S-SPEED_GATE_FULL_RAD_S)));
    }
    float target=gt*gv;
    if(mode==NOTCH_ADAPTIVE) {
      if(target>appliedGamma) holdSeconds=NOTCH_HOLD_S;
      else if(holdSeconds>0) { holdSeconds=fmaxf(0,float(holdSeconds-dt)); target=fmaxf(target,appliedGamma); }
      const float tau=(target>appliedGamma)?NOTCH_ATTACK_S:NOTCH_RELEASE_S;
      appliedGamma+=float(1-exp(-dt/tau))*(target-appliedGamma);
    } else { appliedGamma=target; holdSeconds=0; }
    signals.gammaTheta=gt; signals.gammaSpeed=gv; signals.gamma=appliedGamma;
    signals.notchInput=raw; signals.notchOutput=filtered; signals.notchMode=uint8_t(mode);
    // NOTE: RFOB torque is no longer attenuated at any frequency.
    // OFF/FIXED/ADAPTIVE now select OFF/full/adaptive band-limited VELOCITY damping.
    const float dampingRequested=-DAMPING_GAIN*appliedGamma*oscillationVelocity;
    const float dampingTorque=fmaxf(-DAMPING_TORQUE_LIMIT_NM,
                              fminf(DAMPING_TORQUE_LIMIT_NM,dampingRequested));
    if(!isfinite(dampingTorque)||!finiteState()) { trip(); return 0; }
    if(!enabled||!ready()) { zeroOutput(); return 0; }
    signals.torqueR=Kr*raw;
    signals.torqueF=Kf*friction(signals.velocity,FF_FRICTION_MODEL);
    signals.unsaturated=signals.torqueR+signals.torqueF+dampingTorque;
    if(!finiteState()) { trip(); return 0; }
    signals.command=fmaxf(-TORQUE_LIMIT,fminf(TORQUE_LIMIT,signals.unsaturated));
    signals.saturated=signals.command!=signals.unsaturated;
    if(!isfinite(signals.command)) trip();
    return signals.command;
  }
 private:
  float updateVelocityMovingAverage(float rawVelocity) {
    if(velocityCount<ControlConfig::VELOCITY_MOVING_AVERAGE_SAMPLES) {
      velocitySamples[velocityIndex]=rawVelocity;
      velocitySum+=rawVelocity;
      ++velocityCount;
    } else {
      velocitySum-=velocitySamples[velocityIndex];
      velocitySamples[velocityIndex]=rawVelocity;
      velocitySum+=rawVelocity;
    }
    velocityIndex=uint8_t((velocityIndex+1)%ControlConfig::VELOCITY_MOVING_AVERAGE_SAMPLES);
    return velocitySum/float(velocityCount);
  }
  float velocitySamples[ControlConfig::VELOCITY_MOVING_AVERAGE_SAMPLES]={};
  float velocitySum=0;
  uint8_t velocityIndex=0,velocityCount=0;
  FixedNotch notch,velocityNotch; AngleDetector detector; bool initialized=false,numericFault=false;
  uint32_t lastUs=0; float latestRawVelocity=0;
  ControlConfig::NotchMode mode=ControlConfig::INITIAL_NOTCH_MODE;
  float appliedGamma=0,holdSeconds=0;
  float angleStartThresholdDeg=ControlConfig::ANGLE_NOTCH_START_DEG_RMS;
  float angleFullThresholdDeg=ControlConfig::ANGLE_NOTCH_FULL_DEG_RMS;
};
}

===== control_log.h =====
#pragma once
#include <Arduino.h>

namespace ControlLog {
constexpr uint32_t CAPACITY=16384; // About 8.2 s at 2 kHz; oldest samples are overwritten when full.
struct Sample {
  uint32_t timestampUs=0;
  float position=0, velocity=0;
};
static Sample samples[CAPACITY];
static uint32_t head=0, count=0;

inline void reset() {
  head=0;
  count=0;
}

// Called from the control ISR once for each accepted CAN feedback frame.
inline void append(uint32_t timestampUs,float position,float velocity) {
  samples[head].timestampUs=timestampUs;
  samples[head].position=position;
  samples[head].velocity=velocity;
  head=(head+1)%CAPACITY;
  if(count<CAPACITY) ++count;
}

// Call only while the control session is stopped; serial output is blocking.
inline void dump() {
  Serial.print("LOG_BEGIN,"); Serial.println(count);
  Serial.println("timestamp_us,position_rad,velocity_raw_rad_s");
  const uint32_t oldest=(head+CAPACITY-count)%CAPACITY;
  for(uint32_t i=0;i<count;++i) {
    const Sample &s=samples[(oldest+i)%CAPACITY];
    Serial.print(s.timestampUs); Serial.print(',');
    Serial.print(s.position,7); Serial.print(',');
    Serial.println(s.velocity,7);
  }
  Serial.println("LOG_END");
}
}

#pragma once
#include <math.h>
#include "control_config.h"
#include "notch_coefficients.h"
namespace ControlAlgorithm {
// Asymmetric pole-shift biquad: H=(1-depth)+depth*Hnotch.
// MATLAB fitted coefficients; two states, no added block delay.
// Double states avoid cancellation at low f0/fs. No trigonometry in update().
class FixedNotch {
 public:
  FixedNotch() {
    b0=NotchCoefficients::COEF_B0; b1=NotchCoefficients::COEF_B1; b2=NotchCoefficients::COEF_B2;
    a1=NotchCoefficients::COEF_A1; a2=NotchCoefficients::COEF_A2;
  }
  void reset() { z1=z2=0; initialized=false; }
  bool finiteState() const { return isfinite(z1)&&isfinite(z2)&&isfinite(b0)&&isfinite(b1)&&isfinite(b2)&&isfinite(a1)&&isfinite(a2); }
  float update(float input) {
    const double x=input;
    if(!initialized) { z1=(1-b0)*x; z2=(b2-a2)*x; initialized=true; }
    const double y=b0*x+z1;
    z1=b1*x-a1*y+z2; z2=b2*x-a2*y;
    return float(x+ControlConfig::NOTCH_DEPTH*(y-x));
  }
  double b0=0,b1=0,b2=0,a1=0,a2=0;
 private:
  double z1=0,z2=0;
  bool initialized=false;
};
}

#pragma once
#include <stdint.h>
#include "control_config.h"

namespace PositionDirection {
enum Direction { UNKNOWN, POSITIVE, NEGATIVE };
class Estimator {
 public:
  void reset() { head=0; count=0; result=UNKNOWN; }
  Direction direction() const { return result; }
  void update(float position,uint32_t rxUs) {
    using namespace ControlConfig;
    if(count && rxUs==samples[(head+count-1)%CAPACITY].us) return;
    // Keep the latest received sample at or before now - 50 ms, plus newer samples.
    while(count>1 && uint32_t(rxUs-samples[(head+1)%CAPACITY].us)>=POSITION_DIRECTION_WINDOW_US) {
      head=(head+1)%CAPACITY; --count;
    }
    // Capacity exceeds the number of accepted frames possible in 50 ms at 1 Mbit/s.
    // If exhausted, discard incomplete history and require a fresh full window.
    if(count==CAPACITY) reset();
    samples[(head+count)%CAPACITY]={position,rxUs}; ++count;
    result=UNKNOWN;
    if(uint32_t(rxUs-samples[head].us)<POSITION_DIRECTION_WINDOW_US) return;
    float delta=position-samples[head].position;
    const float range=POSITION_MAX-POSITION_MIN;
    if(delta>range*0.5f) delta-=range;
    else if(delta<-range*0.5f) delta+=range;
    if(delta>POSITION_DIRECTION_THRESHOLD_RAD) result=POSITIVE;
    else if(delta<-POSITION_DIRECTION_THRESHOLD_RAD) result=NEGATIVE;
  }
 private:
  static constexpr uint32_t CAPACITY=1024;
  struct Sample { float position; uint32_t us; };
  Sample samples[CAPACITY];
  uint32_t head=0, count=0;
  Direction result=UNKNOWN;
};
}
